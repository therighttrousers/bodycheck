using BodyCheck.Core.Entities;

namespace BodyCheck.Core.Controllers;

/// <summary>
/// Tuning for a <see cref="PatrolBehaviour"/>. Each value is a function, read whenever it's needed, so a change to the
/// features takes effect at once.
/// </summary>
public sealed record PatrolTuning
{
    /// <summary>The speed below which a patroller trying to walk counts as slow.</summary>
    public required Func<float> StallSpeed { get; init; }
    /// <summary>How long, in game seconds, a patroller must stay slow to count as stalled.</summary>
    public required Func<float> StallTime { get; init; }
    /// <summary>The heat that each stall adds.</summary>
    public required Func<float> StallHeat { get; init; }
    /// <summary>The heat at which a patroller gives up and stands still until it cools down.</summary>
    public required Func<float> StallMaxHeat { get; init; }
    /// <summary>How much heat a patroller loses per game second.</summary>
    public required Func<float> StallCooldownRate { get; init; }

    /// <summary>The tuning for robots.</summary>
    public static PatrolTuning Robot { get; } = new()
    {
        StallSpeed = () => Features.Current.RobotStallSpeed,
        StallTime = () => Features.Current.RobotStallTime,
        StallHeat = () => Features.Current.RobotStallHeat,
        StallMaxHeat = () => Features.Current.RobotStallMaxHeat,
        StallCooldownRate = () => Features.Current.RobotStallCooldownRate,
    };
}

/// <summary>What a patroller means to do.</summary>
/// <param name="Move">The direction to walk in: -1 for left, 1 for right, or 0 to stand still.</param>
/// <param name="Facing">The direction to face: -1 for left, 1 for right, or 0 for neither.</param>
public readonly record struct PatrolIntent(float Move, float Facing);

/// <summary>
/// Makes a walker patrol: walk one way until it stalls, such as against a wall, then turn around and walk the other.
/// </summary>
/// <remarks>
/// A walker that stalls too often, such as one boxed in or held in place, overheats: it stands still until it has
/// cooled down, rather than turning back and forth on the spot.
/// </remarks>
/// <param name="entity">The walker.</param>
/// <param name="tuning">How readily it stalls and overheats.</param>
public sealed class PatrolBehaviour(WalkerEntity entity, PatrolTuning tuning)
{
    // The way to walk when nothing else decides it.
    private const float ArbitraryFirstMove = 1;

    // The direction it's walking in, or 0 if it has yet to choose one.
    private float _move;
    private readonly StallDetector _stall = new(new StallTuning { MinSpeed = tuning.StallSpeed, MaxTime = tuning.StallTime });
    private readonly OverheatDetector _stallHeat = new(new OverheatTuning
    {
        HeatPerEvent = tuning.StallHeat,
        MaxHeat = tuning.StallMaxHeat,
        CooldownRate = tuning.StallCooldownRate,
    });

    /// <summary>Decides what the walker means to do now.</summary>
    /// <param name="patrols">Whether it's patrolling. When it isn't, it stands still, facing as it did.</param>
    /// <param name="facing">The direction it's facing: -1 for left, 1 for right, or 0 for neither.</param>
    /// <param name="gameDt">The game time since the last call.</param>
    /// <returns>The direction to walk in and the direction to face.</returns>
    public PatrolIntent Apply(bool patrols, float facing, float gameDt)
    {
        // It cools down all the time, even when it isn't patrolling.
        _stallHeat.CoolDown(gameDt);

        if (!patrols)
        {
            _move = 0;
            _stall.Reset();
            return new PatrolIntent(0, facing);
        }

        // It walks only on its own feet on the ground: not in the air, or while something else supports it.
        if (!entity.OnGround || entity.Support is not null)
        {
            _move = 0;
            _stall.Reset();
            return new PatrolIntent(0, 0);
        }

        bool contactLeft = entity.Contacts.HasFlag(Contacts.Left);
        bool contactRight = entity.Contacts.HasFlag(Contacts.Right);

        bool stalled = _stall.Update(entity.LinearVelocity.X * _move, gameDt);

        if (stalled)
        {
            _stallHeat.HeatUp();
        }

        if (_stallHeat.Overheated)
        {
            facing = _move = 0;
            _stall.Reset();
        }
        else if (_move == 0)
        {
            // Walk away from whatever it's touching, and not at all if it's touching both sides. Otherwise, walk the
            // way it's facing, or, if it faces neither way, an arbitrary one.
            facing = _move = contactLeft ? (contactRight ? 0 : 1) : contactRight ? -1 : facing != 0 ? facing : ArbitraryFirstMove;
            _stall.Reset();
        }
        else if (_move < 0 && stalled)
        {
            // It stalled walking left, so it turns right, unless something's in the way there too.
            facing = _move = contactRight ? 0 : 1;
            _stall.Reset();
        }
        else if (_move > 0 && stalled)
        {
            // And the same the other way.
            facing = _move = contactLeft ? 0 : -1;
            _stall.Reset();
        }
        return new PatrolIntent(_move, facing);
    }
}
