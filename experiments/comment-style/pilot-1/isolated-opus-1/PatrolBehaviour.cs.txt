using BodyCheck.Core.Entities;

namespace BodyCheck.Core.Controllers;

/// <summary>
/// Tuning for a <see cref="PatrolBehaviour"/>. Each value is a function, read when it is needed, so that it can
/// follow tuning that changes while the game runs.
/// </summary>
public sealed record PatrolTuning
{
    /// <summary>The speed, in the direction of patrol, below which a walker counts as slow.</summary>
    public required Func<float> StallSpeed { get; init; }

    /// <summary>How long, in game seconds, a walker must stay slow to count as stalled, and turn around.</summary>
    public required Func<float> StallTime { get; init; }

    /// <summary>The heat that each stall adds.</summary>
    public required Func<float> StallHeat { get; init; }

    /// <summary>The heat at which a walker that keeps stalling gives up and stands still until it cools.</summary>
    public required Func<float> StallMaxHeat { get; init; }

    /// <summary>How much heat a walker loses per game second.</summary>
    public required Func<float> StallCooldownRate { get; init; }

    /// <summary>The tuning for robots.</summary>
    public static PatrolTuning Robot { get; } = new()
    {
        StallSpeed = () => Features.Current.RobotStallSpeed,
        StallTime = () => Features.Current.RobotStallTime,
        StallHeat = () => Features.Current.RobotStallHeat,
        StallMaxHeat = () => Features.Current.RobotStallMaxHeat,
        StallCooldownRate = () => Features.Current.RobotStallCooldownRate,
    };
}

/// <summary>What a patrolling walker means to do this step.</summary>
/// <param name="Move">The direction to walk: -1 for left, 1 for right, or 0 to stand still.</param>
/// <param name="Facing">The direction to face: -1 for left, 1 for right, or 0 for neither.</param>
public readonly record struct PatrolIntent(float Move, float Facing);

/// <summary>
/// Walks a walker back and forth along the ground, turning around when it stalls, such as against a wall or
/// something it can't push.
/// </summary>
/// <remarks>
/// A walker that keeps stalling, such as one boxed in, heats up, and when it overheats it stands still until it has
/// cooled, rather than turning back and forth on the spot.
/// </remarks>
public sealed class PatrolBehaviour(WalkerEntity entity, PatrolTuning tuning)
{
    // Which way to walk when nothing else decides it.
    private const float ArbitraryFirstMove = 1;

    private float _move;
    private readonly StallDetector _stall = new(new StallTuning { MinSpeed = tuning.StallSpeed, MaxTime = tuning.StallTime });
    private readonly OverheatDetector _stallHeat = new(new OverheatTuning
    {
        HeatPerEvent = tuning.StallHeat,
        MaxHeat = tuning.StallMaxHeat,
        CooldownRate = tuning.StallCooldownRate,
    });

    /// <summary>
    /// Decides what the walker means to do this step.
    /// </summary>
    /// <param name="patrols">Whether the walker is patrolling at all.</param>
    /// <param name="facing">The direction the walker faces now.</param>
    /// <param name="gameDt">The step's length in game time.</param>
    /// <returns>The direction to walk and to face.</returns>
    public PatrolIntent Apply(bool patrols, float facing, float gameDt)
    {
        // Cool down whatever happens, so that a walker that isn't patrolling still recovers.
        _stallHeat.CoolDown(gameDt);

        // A walker that isn't patrolling stands still and keeps facing as it was.
        if (!patrols)
        {
            _move = 0;
            _stall.Reset();
            return new PatrolIntent(0, facing);
        }

        // In the air, or carried on something, the walker can't walk: it stands still, facing neither way, and
        // picks a direction afresh when it lands.
        if (!entity.OnGround || entity.Support is not null)
        {
            _move = 0;
            _stall.Reset();
            return new PatrolIntent(0, 0);
        }

        bool contactLeft = entity.Contacts.HasFlag(Contacts.Left);
        bool contactRight = entity.Contacts.HasFlag(Contacts.Right);

        // Speed in the direction of patrol. Standing still, it's always 0, and so slow, but that is handled below.
        bool stalled = _stall.Update(entity.LinearVelocity.X * _move, gameDt);

        if (stalled)
        {
            _stallHeat.HeatUp();
        }

        if (_stallHeat.Overheated)
        {
            facing = _move = 0;
            _stall.Reset();
        }
        else if (_move == 0)
        {
            // Starting out: walk away from whatever it's touching, and if it's touching both sides, stay put.
            // Otherwise keep the way it faces, or pick one if it faces neither.
            facing = _move = contactLeft ? (contactRight ? 0 : 1) : contactRight ? -1 : facing != 0 ? facing : ArbitraryFirstMove;
            _stall.Reset();
        }
        else if (_move < 0 && stalled)
        {
            // Stalled walking left: turn right, unless something is in the way there too.
            facing = _move = contactRight ? 0 : 1;
            _stall.Reset();
        }
        else if (_move > 0 && stalled)
        {
            // Stalled walking right: turn left, unless something is in the way there too.
            facing = _move = contactLeft ? 0 : -1;
            _stall.Reset();
        }
        return new PatrolIntent(_move, facing);
    }
}
