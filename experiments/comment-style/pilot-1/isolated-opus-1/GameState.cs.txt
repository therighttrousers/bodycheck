using System;
using System.Collections.Generic;
using System.Linq;

using BodyCheck.Core;
using BodyCheck.Core.Entities;
using BodyCheck.Core.WorldData;

using Godot;

namespace BodyCheck.State;

/// <summary>
/// The state of a game in progress: the player, the story, every entity's state in the worlds visited so far, and
/// where to respawn in each. It lasts across worlds, and is what gets saved and loaded.
/// </summary>
public sealed partial class GameState : RefCounted, IGameState
{
    // Every entity's state, by ID, including the player's.
    private readonly Dictionary<string, EntityState> _entities = [];

    // The entities that have names, by untranslated name, including the player.
    private readonly Dictionary<string, EntityState> _entitiesByName = [];

    // The ID of the respawn point in each world, by world name.
    private readonly Dictionary<string, string> _respawnIds = [];

    // Saved data for entities in worlds not yet visited since the game was loaded, by ID. An entity's data moves
    // out of here into its state when its world is filled in, and is saved again as it was if that never happens.
    private readonly Dictionary<string, EntityData> _saved = [];

    /// <summary>Makes the state of a new game, started now, with no worlds in it yet.</summary>
    public GameState() : this(DateTimeOffset.Now, story: null, player: null) { }

    /// <summary>Makes the state of a new game, started now, with <paramref name="world"/>'s entities in it.</summary>
    public GameState(World world) : this() => Fill(world);

    /// <summary>
    /// Makes the state of a game loaded from <paramref name="save"/>. Its entities other than the player get their
    /// states when their worlds are filled in.
    /// </summary>
    public GameState(SaveData save)
        : this(save.Started, save.Story, save.Entities.GetValueOrDefault(Constants.PlayerId) as PlayerData)
    {
        WorldName = save.WorldName;
        foreach ((string worldName, string respawnId) in save.RespawnIds)
        {
            _respawnIds[worldName] = respawnId;
        }
        foreach ((string id, EntityData data) in save.Entities)
        {
            if (id != Constants.PlayerId)
            {
                _saved[id] = data;
            }
        }
    }

    private GameState(DateTimeOffset started, StoryState? story, PlayerData? player)
    {
        Started = started;
        Story = story ?? new();
        Player = new PlayerState(Constants.PlayerId, PlayerKindName, player ?? new());
        _entities[Player.Id] = Player;
        _entitiesByName[Player.UntranslatedName] = Player;
    }

    /// <summary>When the game was started, which stays the same across saves and loads.</summary>
    public DateTimeOffset Started { get; }

    /// <summary>
    /// Makes the data to save this game as, including the saved data of entities in worlds not visited since it was
    /// loaded.
    /// </summary>
    /// <param name="saved">When the game is being saved.</param>
    public SaveData ToSave(DateTimeOffset saved)
    {
        var entities = new Dictionary<string, EntityData>(_saved);
        foreach (EntityState entity in _entities.Values)
        {
            // Only some kinds of entity have state worth saving.
            if (entity.Saved is EntityData data)
            {
                entities[entity.Id] = data;
            }
        }
        return new SaveData
        {
            Version = SaveData.CurrentVersion,
            Started = Started,
            Saved = saved,
            WorldName = WorldName,
            Story = Story,
            RespawnIds = new Dictionary<string, string>(_respawnIds),
            Entities = entities,
        };
    }

    /// <summary>
    /// Gives every entity in <paramref name="world"/> a state, if it doesn't have one yet, from its saved data if
    /// there is any and otherwise from its configuration. Entities that already have states keep them, so that
    /// coming back to a world finds it as it was left.
    /// </summary>
    /// <remarks>
    /// Every entity takes the name it has in <paramref name="world"/>, so that a renamed entity is found by its new
    /// name.
    /// </remarks>
    /// <exception cref="ArgumentException">An entity is named as the player is.</exception>
    public void Fill(World world)
    {
        foreach (EntityConfig config in world.Bodies.Concat(world.Controls))
        {
            if (!_entities.TryGetValue(config.Id, out EntityState? entity))
            {
                var kindName = KindName(config.Kind);
                _saved.Remove(config.Id, out EntityData? saved);
                // Saved data of the wrong kind, such as from an entity whose kind has since changed, is ignored.
                entity = config switch
                {
                    EntityConfig.VerticalDoor => new DoorState(config.Id, kindName, saved as DoorData ?? new()),
                    EntityConfig.Trigger trigger =>
                        new SwitchState(config.Id, kindName, saved as SwitchData ?? new() { IsOn = trigger.StartsOn }),
                    // A once is on until it has happened.
                    EntityConfig.Once => new SwitchState(config.Id, kindName, saved as SwitchData ?? new() { IsOn = true }),
                    _ => new EntityState(config.Id, kindName),
                };
                _entities[config.Id] = entity;
            }
            Rename(entity, config.Name);
        }
    }

    /// <summary>Gives <paramref name="entity"/> the name <paramref name="name"/>, or no name if it's null.</summary>
    /// <exception cref="ArgumentException"><paramref name="name"/> is the player's name.</exception>
    private void Rename(EntityState entity, string? name)
    {
        // Only drop the old name if it is still this entity's: another entity may have taken it since.
        if (entity.GivenName is string old && _entitiesByName.GetValueOrDefault(old) == entity)
        {
            _entitiesByName.Remove(old);
        }
        if (name is not null && _entitiesByName.GetValueOrDefault(name) == Player)
        {
            throw new ArgumentException($"Entity {entity.Id} is named '{name}', as the player is.", nameof(name));
        }
        entity.GivenName = name;
        if (name is not null)
        {
            _entitiesByName[name] = entity;
        }
    }

    /// <summary>The name of the world the player is in, or null before the game has entered one.</summary>
    public string? WorldName { get; set; }

    /// <summary>The state of the story, which dialogue reads and changes.</summary>
    public StoryState Story { get; }

    /// <summary>The ID of the respawn point in the world named <paramref name="worldName"/>, if it has one yet.</summary>
    public string? GetRespawnId(string worldName) => _respawnIds.GetValueOrDefault(worldName);

    /// <summary>Makes the player respawn at <paramref name="respawnId"/> in the world named <paramref name="worldName"/>.</summary>
    public void SetRespawnId(string worldName, string respawnId) => _respawnIds[worldName] = respawnId;

    private static readonly string PlayerKindName = Translatable.Mark("B-CON", KindContext);

    // The name of an entity of the given kind, marked for translation for the kinds whose names are shown.
    // Other kinds go by their kind's name, untranslated.
    private static string KindName(EntityKind kind) => kind switch
    {
        EntityKind.VerticalDoor => Translatable.Mark("Door", KindContext),
        _ => kind.ToString(),
    };

    /// <summary>The translation context for the names of kinds of entity.</summary>
    internal const string KindContext = "kind";

    /// <summary>The translation context for the names given to entities.</summary>
    internal const string NameContext = "entity name";

    /// <summary>Casts <paramref name="entity"/> to <typeparamref name="T"/>, with a helpful message if it isn't one.</summary>
    /// <exception cref="InvalidOperationException">The entity isn't a <typeparamref name="T"/>.</exception>
    private static T CastEntity<T>(object entity, string id) => entity switch
    {
        T value => value,
        var value => throw new InvalidOperationException($"Entity {id} is a {value.GetType().Name}, not a {typeof(T).Name}."),
    };

    /// <summary>The player's state.</summary>
    public PlayerState Player { get; }

    /// <summary>A speaker in dialogue that isn't an entity: a voice the player hears.</summary>
    public CharacterState Voice { get; } = new(Translatable.Mark("Voice", KindContext));

    /// <summary>The state of the entity with the ID <paramref name="id"/>.</summary>
    /// <exception cref="KeyNotFoundException">No entity filled in so far has that ID.</exception>
    public EntityState GetEntity(string id) => _entities[id];

    /// <summary>The state of the entity with the ID <paramref name="id"/>, which must be a <typeparamref name="T"/>.</summary>
    /// <exception cref="KeyNotFoundException">No entity filled in so far has that ID.</exception>
    /// <exception cref="InvalidOperationException">The entity's state isn't a <typeparamref name="T"/>.</exception>
    public T GetEntity<T>(string id) where T : EntityState => CastEntity<T>(GetEntity(id), id);

    /// <summary>The state of the entity with the untranslated name <paramref name="name"/>.</summary>
    /// <exception cref="KeyNotFoundException">No entity filled in so far has that name.</exception>
    public EntityState GetEntityNamed(string name) => _entitiesByName[name];

    /// <summary>
    /// The speaker in dialogue named <paramref name="name"/>, as it is shown: the voice, or the first entity with that
    /// name, or null if there is none.
    /// </summary>
    public CharacterState? GetSpeaker(string name) =>
        Voice.Name == name ? Voice : _entities.Values.FirstOrDefault(entity => entity.Name == name);

    IGameState.IPlayerState IGameState.Player => Player;

    T IGameState.GetEntity<T>(string id) => CastEntity<T>(_entities[id], id);
}
